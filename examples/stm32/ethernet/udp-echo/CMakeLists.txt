set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(sky_ethernet_udp LANGUAGES C ASM)
include("../../common/sky-hal.cmake")

target_include_directories(sky-ethernet-udp.elf PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${lwip}/include")
