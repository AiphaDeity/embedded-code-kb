cmake_minimum_required(VERSION 3.20)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)
set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_CXX_COMPILER arm-none-eabi-g++)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(sky_lvgl_touch LANGUAGES C CXX ASM)
include("../../common/sky-hal.cmake")
add_sky_hal_example(sky-lvgl-touch.elf sram i2c)
target_sources(sky-lvgl-touch.elf PRIVATE "${CUBE_F4_ROOT}/Drivers/STM32F4xx_HAL_Driver/Src/stm32f4xx_ll_fsmc.c")

set(LVGL_ROOT "" CACHE PATH "Path to an LVGL 9.6 source tree")
if(NOT EXISTS "${LVGL_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "Set LVGL_ROOT to an LVGL 9.6 source tree.")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h")
set(CONFIG_LV_BUILD_EXAMPLES OFF)
set(CONFIG_LV_BUILD_DEMOS OFF)
set(CONFIG_LV_USE_THORVG OFF)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF)
set(LV_FETCH_DEPENDENCIES OFF)
set(LV_BUILD_INSTALL OFF)
add_subdirectory("${LVGL_ROOT}" lvgl EXCLUDE_FROM_ALL)
target_compile_options(lvgl PRIVATE -mcpu=cortex-m4 -mthumb -mfpu=fpv4-sp-d16 -mfloat-abi=hard
  -Os -ffunction-sections -fdata-sections)
target_link_libraries(sky-lvgl-touch.elf PRIVATE lvgl::lvgl m)
