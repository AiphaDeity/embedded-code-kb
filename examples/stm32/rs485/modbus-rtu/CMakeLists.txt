cmake_minimum_required(VERSION 3.20)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)
set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(sky_modbus_rtu LANGUAGES C ASM)
include("../../common/sky-hal.cmake")
add_sky_hal_example(sky-modbus-client.elf uart)
add_sky_hal_example(sky-modbus-server.elf uart)
target_compile_definitions(sky-modbus-client.elf PRIVATE RTU_SERVER=0)
target_compile_definitions(sky-modbus-server.elf PRIVATE RTU_SERVER=1)
