cmake_minimum_required(VERSION 3.20)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)
set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(sky_rcc_hse_pll LANGUAGES C ASM)
include("../../common/sky-hal.cmake")
add_sky_hal_example(sky-rcc-hse-pll.elf)
option(SKY_MCO_PLL "Output PLL/4 (12 MHz) on PA8 instead of HSE (8 MHz)" OFF)
target_compile_definitions(sky-rcc-hse-pll.elf PRIVATE HSE_VALUE=8000000U)
if(SKY_MCO_PLL)
  target_compile_definitions(sky-rcc-hse-pll.elf PRIVATE SKY_MCO_PLL=1)
endif()
